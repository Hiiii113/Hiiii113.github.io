const fs = require("fs"); const path = require("path"); const dir = process.cwd(); // 备份目录 const backupDir = path.join( path.dirname(dir), "_posts_backup_" + new Date().toISOString().replace(/[-:T]/g, "").slice(0, 14) ); fs.cpSync(dir, backupDir, { recursive: true }); function walk(folder) { const files = fs.readdirSync(folder); for (const file of files) { const fullPath = path.join(folder, file); const stat = fs.statSync(fullPath); if (stat.isDirectory()) { walk(fullPath); continue; } if (!file.endsWith(".md")) { continue; } let content = fs.readFileSync(fullPath, "utf8"); // 已经有 front-matter 的跳过 if (/^\s*---\s*\r?\n/.test(content)) { console.log("跳过：" + file); continue; } let title = path.basename(file, ".md"); const match = content.match(/^#\s+(.+)$/m); if (match) { title = match[1].trim(); } // 处理 YAML 单引号 title = title.replace(/'/g, "''"); const created = stat.birthtime; const updated = stat.mtime; function formatDate(date) { const pad = (n) => String(n).padStart(2, "0"); return ( date.getFullYear() + "-" + pad(date.getMonth() + 1) + "-" + pad(date.getDate()) + " " + pad(date.getHours()) + ":" + pad(date.getMinutes()) + ":" + pad(date.getSeconds()) ); } const frontmatter = "---\n" + "title: '" + title + "'\n" + "date: " + formatDate(created) + "\n" + "updated: " + formatDate(updated) + "\n" + "tags:\n" + "categories:\n" + "---\n\n"; fs.writeFileSync(fullPath, frontmatter + content, "utf8"); console.log("已添加：" + file); } } walk(dir); console.log("处理完成"); console.log("备份目录：" + backupDir);
